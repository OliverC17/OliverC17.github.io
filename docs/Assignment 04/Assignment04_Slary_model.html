def calculate_bic_from_glr(summary, k, n):
    deviance = summary.deviance
    dispersion = summary.dispersion
    
    log_likelihood = -0.5 * (
        n * math.log(2 * math.pi) +
        n * math.log(dispersion) +
        (deviance / dispersion)
    )
    
    bic = k * math.log(n) - 2 * log_likelihood
    return log_likelihood, bic
